package rpcclient // import "github.com/anthony-c-martin/bicep-testing/packages/go/rpcclient"

Package rpcclient installs and communicates with the Bicep CLI over JSON-RPC.

FUNCTIONS

func DownloadURL(ctx context.Context, version string) (string, error)
    DownloadURL returns the Bicep CLI download URL for a version and the current
    platform.

func Install(ctx context.Context, basePath, version string) (string, error)
    Install downloads the Bicep CLI into basePath and returns its executable
    path.


TYPES

type Client struct {
	// Has unexported fields.
}
    Client communicates with one Bicep CLI process. Calls on a Client are safe
    for concurrent use.

func New(ctx context.Context, bicepPath string) (*Client, error)
    New starts the Bicep CLI at bicepPath and validates its RPC protocol
    version.

func (client *Client) Close() error
    Close disconnects from the Bicep CLI and terminates its process.

func (client *Client) CompileParams(ctx context.Context, request CompileParamsRequest) (CompileParamsResponse, error)
    CompileParams compiles a Bicep parameters file into deployable ARM JSON.

func (client *Client) GetSnapshot(ctx context.Context, request GetSnapshotRequest) (GetSnapshotResponse, error)
    GetSnapshot returns a deployment snapshot for a Bicep parameters file.

func (client *Client) Version(ctx context.Context) (string, error)
    Version returns the Bicep CLI version.

type CompileParamsRequest struct {
	Path               string                     `json:"path"`
	ParameterOverrides map[string]json.RawMessage `json:"parameterOverrides"`
}
    CompileParamsRequest contains the path and overrides for a Bicep parameters
    compilation.

type CompileParamsResponse struct {
	Success     bool             `json:"success"`
	Diagnostics []map[string]any `json:"diagnostics"`
	Template    string           `json:"template"`
	Parameters  string           `json:"parameters"`
}
    CompileParamsResponse contains deployable ARM template and parameter JSON.

type GetSnapshotRequest struct {
	Path           string                  `json:"path"`
	Metadata       SnapshotMetadata        `json:"metadata"`
	ExternalInputs []SnapshotExternalInput `json:"externalInputs,omitempty"`
}
    GetSnapshotRequest contains the path and deployment context for a Bicep
    snapshot.

type GetSnapshotResponse struct {
	Snapshot string `json:"snapshot"`
}
    GetSnapshotResponse contains the serialized deployment snapshot.

type RPCError struct {
	Code    int             `json:"code"`
	Message string          `json:"message"`
	Data    json.RawMessage `json:"data,omitempty"`
}
    RPCError is an error returned by the Bicep JSON-RPC server.

func (err *RPCError) Error() string

type SnapshotExternalInput struct {
	Kind   string `json:"kind"`
	Config any    `json:"config,omitempty"`
	Value  any    `json:"value"`
}
    SnapshotExternalInput supplies a value for a Bicep external input.

type SnapshotMetadata struct {
	TenantID       string `json:"tenantId,omitempty"`
	SubscriptionID string `json:"subscriptionId,omitempty"`
	ResourceGroup  string `json:"resourceGroup,omitempty"`
	Location       string `json:"location,omitempty"`
	DeploymentName string `json:"deploymentName,omitempty"`
}
    SnapshotMetadata describes the Azure deployment context used to evaluate a
    snapshot.
