package biceptesting // import "github.com/anthony-c-martin/bicep-testing/packages/go/bicep-testing"

Package biceptesting provides helpers for testing Bicep infrastructure code.

TYPES

type DeployOptions struct {
	FilePath           string
	ManagementGroupID  string
	SubscriptionID     string
	ResourceGroup      string
	Location           string
	StackName          string
	ParameterOverrides map[string]json.RawMessage
}
    DeployOptions identifies a Bicep parameters file and its Deployment Stack
    scope.

type DeployResult struct {
	Succeeded    bool
	Error        *OperationError
	ErrorCode    string
	ErrorMessage string
	Outputs      map[string]any
	Resources    []DeploymentResource

	// Has unexported fields.
}
    DeployResult contains deployment status, outputs, resources, and cleanup
    ownership.

func (result *DeployResult) Teardown(ctx context.Context) error
    Teardown deletes the Deployment Stack and all resources it manages.
    Concurrent calls share an active deletion, and a failed deletion can be
    retried.

type DeploymentResource struct {
	ID   string
	Type string
}
    DeploymentResource identifies a resource managed by a Deployment Stack.

type LiveSession struct {
	// Has unexported fields.
}
    LiveSession owns a credential and an offline session for snapshot,
    validation, and deployment.

func NewLiveSession(ctx context.Context, bicepVersion string, credential azcore.TokenCredential) (*LiveSession, error)
    NewLiveSession creates a credential-owning live session.

func (session *LiveSession) Close() error
    Close disconnects from the Bicep CLI and terminates its process.

func (session *LiveSession) Deploy(ctx context.Context, options DeployOptions) (*DeployResult, error)
    Deploy compiles and deploys a Bicep parameters file as a Deployment Stack.
    Pre-submission failures return an error. Post-submission Azure failures
    return a failed DeployResult.

func (session *LiveSession) Snapshot(ctx context.Context, filePath string, metadata SnapshotMetadata) (*SnapshotResult, error)
    Snapshot forwards to the owned offline session.

func (session *LiveSession) Validate(ctx context.Context, options DeployOptions) (*ValidateResult, error)
    Validate compiles and validates a Bicep parameters file as a Deployment
    Stack.

type OperationError struct {
	Code    string
	Message string
	RawData json.RawMessage
}
    OperationError captures an Azure operation failure.

type Session struct {
	// Has unexported fields.
}
    Session owns a pinned Bicep CLI used to evaluate infrastructure under test.

func NewSession(ctx context.Context, bicepVersion string) (*Session, error)
    NewSession installs the requested Bicep CLI version if needed and starts its
    RPC client.

func (session *Session) Close() error
    Close disconnects from the Bicep CLI and terminates its process.

func (session *Session) Snapshot(
	ctx context.Context,
	filePath string,
	metadata SnapshotMetadata,
) (*SnapshotResult, error)
    Snapshot evaluates a Bicep parameters file without deploying it.

type SnapshotMetadata = biceprpcclient.SnapshotMetadata
    SnapshotMetadata describes the Azure deployment context used to evaluate a
    snapshot.

type SnapshotResource struct {
	ID                   string         `json:"id"`
	Type                 string         `json:"type"`
	Name                 string         `json:"name"`
	APIVersion           string         `json:"apiVersion"`
	Location             string         `json:"location,omitempty"`
	Properties           map[string]any `json:"properties,omitempty"`
	AdditionalProperties map[string]any `json:"-"`
}
    SnapshotResource is a resource predicted by a Bicep snapshot.

func (resource *SnapshotResource) UnmarshalJSON(data []byte) error
    UnmarshalJSON preserves resource fields that are not part of the common
    snapshot model.

type SnapshotResult struct {
	PredictedResources []SnapshotResource `json:"predictedResources"`
	Diagnostics        []string           `json:"diagnostics"`
	Outputs            map[string]any     `json:"outputs"`
}
    SnapshotResult is the predicted result of evaluating a Bicep parameters
    file.

type ValidateResult struct {
	IsValid       bool
	Resources     []DeploymentResource
	CorrelationID string
	Error         *OperationError
	ErrorCode     string
	ErrorMessage  string
}
    ValidateResult describes a Deployment Stack validation response.
